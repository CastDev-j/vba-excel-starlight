Las estructuras de control permiten que tu programa tome decisiones y repita acciones automáticamente.

## If - Tomar Decisiones

### If Básico

<Code
  code={`Sub IfBasico()
    Dim nota As Integer
    nota = Range("A1").Value
    
    ' Condición simple
    If nota >= 70 Then
        Range("B1").Value = "Aprobado"
    End If
    
    ' Con Else
    If nota >= 70 Then
        Range("C1").Value = "Aprobado"
    Else
        Range("C1").Value = "Reprobado"
    End If
End Sub`}
  lang="VBA"
  title="If simple y con Else"
/>

### If con Múltiples Condiciones

<Code
  code={`Sub IfMultiple()
    Dim calificacion As Integer
    calificacion = Range("A1").Value
    
    If calificacion >= 90 Then
        Range("B1").Value = "A - Excelente"
    ElseIf calificacion >= 80 Then
        Range("B1").Value = "B - Muy Bueno"
    ElseIf calificacion >= 70 Then
        Range("B1").Value = "C - Bueno"
    ElseIf calificacion >= 60 Then
        Range("B1").Value = "D - Suficiente"
    Else
        Range("B1").Value = "F - Insuficiente"
    End If
End Sub`}
  lang="VBA"
  title="If con ElseIf"
/>

## For - Repetir Acciones

### For Básico

<Code
  code={`Sub ForBasico()
    Dim i As Integer
    
    ' Contar del 1 al 10
    For i = 1 To 10
        Range("A" & i).Value = "Fila " & i
    Next i
    
    ' Con incremento personalizado
    For i = 2 To 20 Step 2  ' Solo pares
        Range("B" & i).Value = i
    Next i
    
    ' Countdown
    For i = 10 To 1 Step -1
        Range("C" & i).Value = "Quedan " & i
    Next i
End Sub`}
  lang="VBA"
  title="Bucles For"
/>

### For Each - Iterar Colecciones

<Code
  code={`Sub ForEach()
    Dim celda As Range
    Dim hoja As Worksheet
    
    ' Iterar rango de celdas
    For Each celda In Range("A1:A10")
        If celda.Value > 100 Then
            celda.Interior.Color = RGB(255, 0, 0)  ' Rojo
        End If
    Next celda
    
    ' Iterar hojas del workbook
    For Each hoja In ThisWorkbook.Worksheets
        hoja.Range("A1").Value = "Procesada: " & Now
    Next hoja
End Sub`}
  lang="VBA"
  title="For Each para colecciones"
/>

## While y Do - Bucles Condicionales

### Do While

<Code
  code={`Sub DoWhile()
    Dim fila As Integer
    fila = 1
    
    ' Ejecutar mientras la condición sea verdadera
    Do While Range("A" & fila).Value <> ""
        Range("B" & fila).Value = "Procesado"
        fila = fila + 1
        
        ' Seguridad para evitar bucle infinito
        If fila > 1000 Then Exit Do
    Loop
    
    Range("C1").Value = "Filas procesadas: " & (fila - 1)
End Sub`}
  lang="VBA"
  title="Do While"
/>

### Do Until

<Code
  code={`Sub DoUntil()
    Dim suma As Double
    Dim fila As Integer
    fila = 1
    
    ' Ejecutar hasta que la condición sea verdadera
    Do Until suma > 1000
        suma = suma + Range("A" & fila).Value
        fila = fila + 1
        
        ' Salir si no hay más datos
        If Range("A" & fila).Value = "" Then Exit Do
    Loop
    
    Range("B1").Value = "Suma total: " & suma
End Sub`}
  lang="VBA"
  title="Do Until"
/>

## Select Case - Múltiples Opciones

<Code
  code={`Sub SelectCase()
    Dim categoria As String
    Dim descuento As Double
    
    categoria = Range("A1").Value
    
    Select Case UCase(categoria)
        Case "VIP"
            descuento = 0.2     ' 20%
        Case "PREMIUM"
            descuento = 0.15    ' 15%
        Case "REGULAR"
            descuento = 0.05    ' 5%
        Case "NUEVO"
            descuento = 0.1     ' 10%
        Case Else
            descuento = 0       ' Sin descuento
    End Select
    
    Range("B1").Value = "Descuento: " & (descuento * 100) & "%"
End Sub`}
  lang="VBA"
  title="Select Case básico"
/>

### Select Case Avanzado

<Code
  code={`Sub SelectCaseAvanzado()
    Dim puntuacion As Integer
    Dim mensaje As String
    
    puntuacion = Range("A1").Value
    
    Select Case puntuacion
        Case Is >= 90
            mensaje = "Sobresaliente"
        Case 80 To 89
            mensaje = "Notable"
        Case 70 To 79
            mensaje = "Bien"
        Case 60 To 69
            mensaje = "Suficiente"
        Case Is < 60
            mensaje = "Insuficiente"
        Case Else
            mensaje = "Valor no válido"
    End Select
    
    Range("B1").Value = mensaje
End Sub`}
  lang="VBA"
  title="Select Case con rangos"
/>

## Exit y Break

<Code
  code={`Sub ControlFlujo()
    Dim i As Integer
    
    ' Exit For - Salir del bucle
    For i = 1 To 100
        If Range("A" & i).Value = "STOP" Then
            Exit For  ' Salir inmediatamente
        End If
        Range("B" & i).Value = "Procesado"
    Next i
    
    ' Exit Sub - Salir del procedimiento
    If Range("C1").Value = "" Then
        MsgBox "No hay datos para procesar"
        Exit Sub  ' Terminar el procedimiento aquí
    End If
    
    ' Resto del código...
End Sub`}
  lang="VBA"
  title="Salir de bucles y procedimientos"
/>

## Ejemplo Práctico Completo

Procesador de ventas con múltiples estructuras:

<Code
  code={`Sub ProcesadorVentas()
    Dim fila As Integer, ultimaFila As Integer
    Dim total As Double, ventaActual As Double
    Dim vendedor As String, region As String
    Dim bonificacion As Double, categoria As String
    
    ' Encontrar última fila con datos
    ultimaFila = Cells(Rows.Count, "A").End(xlUp).Row
    
    ' Validar que hay datos
    If ultimaFila < 2 Then
        MsgBox "No hay datos para procesar"
        Exit Sub
    End If
    
    ' Procesar cada fila de ventas
    For fila = 2 To ultimaFila
        vendedor = Range("A" & fila).Value
        ventaActual = Range("B" & fila).Value
        region = Range("C" & fila).Value
        
        ' Saltar si no hay vendedor
        If vendedor = "" Then
            Range("E" & fila).Value = "ERROR: Sin vendedor"
            GoTo SiguienteFila
        End If
        
        ' Calcular bonificación por región
        Select Case UCase(region)
            Case "NORTE"
                bonificacion = ventaActual * 0.08
            Case "SUR"
                bonificacion = ventaActual * 0.06
            Case "CENTRO"
                bonificacion = ventaActual * 0.07
            Case Else
                bonificacion = ventaActual * 0.05
        End Select
        
        ' Categorizar vendedor por performance
        If ventaActual >= 50000 Then
            categoria = "ESTRELLA"
            bonificacion = bonificacion * 1.5  ' Bonus extra
        ElseIf ventaActual >= 30000 Then
            categoria = "SENIOR"
            bonificacion = bonificacion * 1.2
        ElseIf ventaActual >= 15000 Then
            categoria = "JUNIOR"
        Else
            categoria = "NUEVO"
            bonificacion = bonificacion * 0.8  ' Reducir bonificación
        End If
        
        ' Escribir resultados
        Range("D" & fila).Value = bonificacion
        Range("E" & fila).Value = categoria
        
        total = total + ventaActual
        
SiguienteFila:
    Next fila
    
    ' Resumen final
    Range("F1").Value = "Total procesado: $" & total
    Range("F2").Value = "Filas procesadas: " & (ultimaFila - 1)
    Range("F3").Value = "Fecha: " & Now
    
    MsgBox "Procesamiento completado. Total: $" & total
End Sub`}
  lang="VBA"
  title="Ejemplo completo - Procesador de ventas"
/>

## Estructuras Anidadas

<Code
  code={`Sub EstructurasAnidadas()
    Dim departamento As String, antiguedad As Integer
    Dim salario As Double, aumento As Double
    
    departamento = Range("A1").Value
    antiguedad = Range("B1").Value
    salario = Range("C1").Value
    
    ' If anidado dentro de Select Case
    Select Case UCase(departamento)
        Case "VENTAS"
            If antiguedad >= 5 Then
                aumento = salario * 0.15
            ElseIf antiguedad >= 2 Then
                aumento = salario * 0.10
            Else
                aumento = salario * 0.05
            End If
            
        Case "SISTEMAS"
            If salario < 25000 Then
                aumento = salario * 0.20  ' Ajuste salarial
            Else
                aumento = salario * 0.12
            End If
            
        Case Else
            aumento = salario * 0.08  ' Aumento estándar
    End Select
    
    Range("D1").Value = aumento
End Sub`}
  lang="VBA"
  title="Estructuras anidadas"
/>
